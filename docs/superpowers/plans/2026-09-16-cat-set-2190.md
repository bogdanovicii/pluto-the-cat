> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add the approved five-piece cat set—Spray Bottle, Feather Teaser, Toilet Paper Roll, Cone of Shame and Coffee Mug—complete with approved art, mechanics, synergies, tests, documentation and a packaged 2.19.0 test build.

**Architecture:** Engine-free timing, geometry, hit-count and duration decisions live in a new `CatSetRules.cs`, tested through the existing Mono harness. **Tech Stack:** C# net35 (BepInEx 5, Alexandria 0.5.10, Mono msbuild), Python 3 + Pillow art tooling, Mono `mcs`/`mono` cases, pluto-artist and pluto-pixel-art skills.

**Source of truth:** `docs/superpowers/specs/2026-09-16-cat-japan-items-yasupen-design.md`, especially Shared implementation rules, Round 2, Chosen concepts, Testing and acceptance, and Risks. Do not change its balance numbers or redesign its art.

```csharp
internal static class CatSetRules
{
    public static bool RollFlinch(float roll, float chance) => !float.IsNaN(roll) && roll >= 0f && roll < Math.Max(0f, Math.Min(1f, chance));
    public static float DistractSeconds(bool boss, float normalSeconds, float bossSeconds) => Math.Max(0f, boss ? bossSeconds : normalSeconds);
    public static bool StreamerAlive(float now, float born, float seconds, int hits, int maxHits) => hits < maxHits && now - born < seconds;
    public static bool ConeReady(float now, float lastBlock, float cooldown) => now - lastBlock >= cooldown;
    public static bool InCone(float dx, float dy, float aimX, float aimY, float radius, float degrees)
    {
        if (float.IsNaN(dx) || float.IsNaN(dy) || float.IsNaN(aimX) || float.IsNaN(aimY) || radius <= 0f || degrees <= 0f) return false;
        double d2 = dx * dx + dy * dy, a2 = aimX * aimX + aimY * aimY;
        if (d2 <= 0.0 || a2 <= 0.0 || d2 > radius * radius) return false;
        double dot = (dx * aimX + dy * aimY) / Math.Sqrt(d2 * a2);
        dot = Math.Max(-1.0, Math.Min(1.0, dot));
        return Math.Acos(dot) * 180.0 / Math.PI <= degrees * 0.5;
    }
    public static float ShardAngle(int index, int count) => count <= 0 ? 0f : 360f * index / count;
    public static float ExtendDuration(float duration, float bonus) => Math.Max(0f, duration) + Math.Max(0f, bonus);
    public static float ExtendRemaining(float remaining, float bonus, bool active) => active ? Math.Max(0f, remaining) + Math.Max(0f, bonus) : remaining;
}
```
